server {
    listen 80;
    server_name sleepysoft.org;

    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name sleepysoft.org;

    ssl_certificate /etc/letsencrypt/live/sleepysoft.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/sleepysoft.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    location = /ddns/update {
        proxy_pass http://127.0.0.1:9000/ddns/update$is_args$args;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location /administrator-manager-api {
        if ($http_upgrade != "websocket") { return 404; }
        proxy_redirect off;
        proxy_pass http://127.0.0.1:10000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    error_page 502 503 504 =503 /__site_announcement.html;
    location = /__site_announcement.html {
        internal;
        root /var/www/site-announcement;
        add_header Cache-Control "no-store" always;
        add_header Retry-After "300" always;
    }

    include /etc/nginx/site-modes/active.conf;
}
